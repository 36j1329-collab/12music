<script>
const $ = id => document.getElementById(id);
let matrix = [], rowTimers = [], activeNodes = [], playing = false, audioCtx = null, runToken = 0;

function parseSeries() {
  const raw = $('series').value.trim();
  const nums = raw.split(/[,\s]+/).filter(Boolean).map(Number);
  if (nums.length !== 12 || nums.some(n => !Number.isInteger(n) || n < 0 || n > 11)) {
    throw new Error('0부터 11까지의 정수 12개를 입력해 주세요.');
  }
  if (new Set(nums).size !== 12) throw new Error('중복된 숫자가 있습니다. 0~11을 각각 한 번씩 입력해 주세요.');
  return nums;
}
const mod12 = n => ((n % 12) + 12) % 12;

function makeData() {
  const p = parseSeries();
  // 일반적인 반행: 첫 음을 축으로 음정 간격의 부호를 반전
  const inv = p.map(x => mod12(2 * p[0] - x));
  matrix = inv.map(iv => p.map(pj => mod12(iv + pj - p[0])));
  // 위 식은 첫 행이 P, 첫 열이 I가 되도록 첫 음을 기준으로 정규화한다.
  // 첫 음이 0인 경우 사용자가 제시한 (I_i + P_j) mod 12와 동일하다.
  renderMatrix(p, inv);
  $('inversion').textContent = inv.join(' · ');
  $('status').textContent = '음렬표를 만들었습니다. 각 행과 열에 0~11이 한 번씩 나타나는지 확인할 수 있습니다.';
  return {p, inv};
}
function renderMatrix(p, inv) {
  let html = '<table aria-label="12 곱하기 12 음렬표"><thead><tr><th>행\\열</th>';
  for(let j=0;j<12;j++) html += `<th>${j+1}</th>`;
  html += '</tr></thead><tbody>';
  matrix.forEach((row,i) => {
    html += `<tr><th>${i+1}</th>`;
    row.forEach((v,j) => html += `<td class="${i===0||j===0?'head':''}" id="cell-${i}-${j}">${v}</td>`);
    html += '</tr>';
  });
  html += '</tbody></table>';
  $('matrix').innerHTML = html;
}
function clearHighlights() {
  document.querySelectorAll('td.active').forEach(el => el.classList.remove('active'));
}
function highlightPair(n, step) {
  clearHighlights();
  // 현재 n행의 step번째 음과 n열의 step번째 음
  const a = $(`cell-${n}-${step}`), b = $(`cell-${step}-${n}`);
  if(a) a.classList.add('active');
  if(b) b.classList.add('active');
}
function midiFreq(note) { return 440 * Math.pow(2, (note - 69) / 12); }
function playTone(freq, start, duration, wave, gain=0.11) {
  const osc = audioCtx.createOscillator();
  const amp = audioCtx.createGain();
  osc.type = wave;
  osc.frequency.value = freq;
  amp.gain.setValueAtTime(0.0001, start);
  amp.gain.exponentialRampToValueAtTime(gain, start + 0.025);
  amp.gain.setTargetAtTime(0.0001, start + Math.max(0.04, duration - 0.04), 0.035);
  osc.connect(amp); amp.connect(audioCtx.destination);
  osc.start(start); osc.stop(start + duration + 0.08);
  activeNodes.push(osc);
}
async function playMusic() {
  let p;
  try { ({p} = makeData()); } catch(e) { $('status').textContent = e.message; return; }
  stopMusic(false);
  const token = ++runToken;
  audioCtx = new (window.AudioContext || window.webkitAudioContext)();
  await audioCtx.resume();
  const bpm = Math.max(40, Math.min(200, Number($('bpm').value) || 100));
  const beat = 60 / bpm, noteDur = beat * Number($('length').value);
  const wave = $('wave').value, startAt = audioCtx.currentTime + 0.12;
  playing = true; $('play').disabled = true; $('stop').disabled = false;
  $('status').textContent = '재생 중: 같은 번호의 행(높은 성부)과 열(낮은 성부)을 짝지어 연주합니다.';
  for(let n=0;n<12;n++) {
    if(token !== runToken) return;
    for(let k=0;k<12;k++) {
      const t = startAt + (n*12+k)*beat;
      const upper = matrix[n][k];       // 행: 높은 성부
      const lower = matrix[k][n];       // 열: 낮은 성부
      playTone(midiFreq(60 + upper), t, noteDur, wave, 0.095);
      playTone(midiFreq(48 + lower), t, noteDur, wave, 0.075);
      const delay = Math.max(0, (t - audioCtx.currentTime) * 1000);
      const timer = setTimeout(() => { if(token === runToken) highlightPair(n,k); }, delay);
      rowTimers.push(timer);
    }
  }
  const totalMs = 12*12*beat*1000 + 300;
  setTimeout(() => {
    if(token === runToken) {
      playing = false; $('play').disabled = false; $('stop').disabled = true;
      clearHighlights(); $('status').textContent = '재생이 끝났습니다. 음렬표의 행과 열이 두 성부로 구현되었습니다.';
      if(audioCtx) { audioCtx.close(); audioCtx = null; }
      activeNodes = [];
    }
  }, totalMs);
}
function stopMusic(update=true) {
  runToken++;
  rowTimers.forEach(clearTimeout); rowTimers = [];
  activeNodes.forEach(node => { try { node.stop(); } catch(e) {} });
  activeNodes = [];
  if(audioCtx) { audioCtx.close(); audioCtx = null; }
  playing = false; $('play').disabled = false; $('stop').disabled = true;
  clearHighlights();
  if(update) $('status').textContent = '재생을 정지했습니다.';
}
$('build').addEventListener('click', () => { stopMusic(false); try { makeData(); } catch(e) { $('status').textContent=e.message; } });
$('play').addEventListener('click', playMusic);
$('stop').addEventListener('click', () => stopMusic(true));
makeData();
</script>
